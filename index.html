// sam.raw.js — RAW
// SAM sagt:  ZIEL + RICHTUNG + SCHRITT
// OUR sagt:  VERSTANDEN + GEPRÜFT + BEREIT
// GPU macht: ZEICHNEN + RECHNEN + HALTEN
const SAM = { say:(ziel,richtung,schritt)=>{
  if(!ziel||!richtung||schritt===undefined) throw new Error("SAM braucht ZIEL + RICHTUNG + SCHRITT");
  return {ziel,richtung,schritt, von:"SAM", an:"OUR"};
}};
const OUR = { pruefen:(s)=>/^[1-9]-[1-9]-[1-9]$/.test(s.ziel) && ["X","Y","Z","45","90"].includes(String(s.richtung)) && Number.isInteger(s.schritt) };
function schritt(ziel,richtung,schrittWeite){
  const satz = SAM.say(ziel,richtung,schrittWeite);
  const ok = OUR.pruefen(satz);
  return ok ? "VERSTANDEN + GEPRÜFT + BEREIT — GPU: ZEICHNEN + RECHNEN + HALTEN" : "NICHT BEREIT — erneut prüfen";
}
// Test im Kopf: schritt("5-5-5","Z",81)
